   100→   100→-- Auto-create a profile row when a new auth.users row is created
   101→   101→CREATE OR REPLACE FUNCTION public.handle_new_user()
   102→   102→RETURNS TRIGGER AS $$
   103→   103→DECLARE
   104→   104→  v_role TEXT;
   105→   105→  v_name TEXT;
   106→   106→BEGIN
   107→   107→  -- Read role/name from raw_user_meta_data, defaulting safely
   108→   108→  v_role := COALESCE(NEW.raw_user_meta_data->>'role', 'LEARNER');
   109→   109→  IF v_role NOT IN ('ADMIN', 'TEACHER', 'LEARNER') THEN
   110→   110→    v_role := 'LEARNER';
   111→   111→  END IF;
   112→   112→  v_name := COALESCE(NEW.raw_user_meta_data->>'name', split_part(NEW.email, '@', 1));
   113→   113→  INSERT INTO public.profiles (id, email, name, role)
   114→   114→  VALUES (NEW.id, NEW.email, v_name, v_role::user_role)
   115→   115→  ON CONFLICT (id) DO NOTHING;
   116→   116→  RETURN NEW;
   117→   117→END;
   118→   118→$$ LANGUAGE plpgsql SECURITY DEFINER;
